#!/bin/sh

cd /var/www/html

. /etc/s6-overlay/scripts/container-role
role="$(tinyhelm_container_role_value)"

if ! tinyhelm_container_has_role flux; then
    echo "   INFO  Flux is disabled for role '$role', sleeping."
    exec sleep infinity
fi

if grep -qE '^TINYHELM_FLUX_ENABLED=false' .env 2>/dev/null || [ "${TINYHELM_FLUX_ENABLED:-}" = "false" ]; then
    echo "   INFO  Flux is disabled, sleeping."
    exec sleep infinity
fi

export TINYHELM_FLUX_GRPC_BIND="${TINYHELM_FLUX_GRPC_BIND:-0.0.0.0:6443}"
export TINYHELM_FLUX_UNIX_SOCKET_PATH="${TINYHELM_FLUX_UNIX_SOCKET_PATH:-/run/tinyhelm/flux.sock}"
export TINYHELM_FLUX_JWT_PRIVATE_KEY_PATH="${TINYHELM_FLUX_JWT_PRIVATE_KEY_PATH:-/var/www/html/storage/app/flux/jwt.priv}"
export TINYHELM_FLUX_JWT_PUBLIC_KEY_PATH="${TINYHELM_FLUX_JWT_PUBLIC_KEY_PATH:-/var/www/html/storage/app/flux/jwt.pub}"
export TINYHELM_FLUX_ALLOW_PUBLIC_BIND="${TINYHELM_FLUX_ALLOW_PUBLIC_BIND:-1}"
export TINYHELM_FLUX_REQUIRE_HOST_BINDING="${TINYHELM_FLUX_REQUIRE_HOST_BINDING:-0}"
export TINYHELM_FLUX_LARAVEL_API_URL="${TINYHELM_FLUX_LARAVEL_API_URL:-http://127.0.0.1:8080}"
if [ -z "${TINYHELM_FLUX_LARAVEL_API_TOKEN:-}" ] && [ -f .env ]; then
    TINYHELM_FLUX_LARAVEL_API_TOKEN="$(grep -E '^TINYHELM_FLUX_LARAVEL_API_TOKEN=' .env 2>/dev/null | tail -n1 | cut -d= -f2- | sed "s/^['\"]//; s/['\"]$//")"
fi
export TINYHELM_FLUX_LARAVEL_API_TOKEN="${TINYHELM_FLUX_LARAVEL_API_TOKEN:-}"
if [ ! -r "$TINYHELM_FLUX_JWT_PUBLIC_KEY_PATH" ]; then
    echo "   INFO  Flux JWT public key not found at $TINYHELM_FLUX_JWT_PUBLIC_KEY_PATH, generating keypair..."
    mkdir -p "$(dirname "$TINYHELM_FLUX_JWT_PRIVATE_KEY_PATH")" "$(dirname "$TINYHELM_FLUX_JWT_PUBLIC_KEY_PATH")"
    openssl genpkey -algorithm EC -pkeyopt ec_paramgen_curve:P-256 -out "$TINYHELM_FLUX_JWT_PRIVATE_KEY_PATH.tmp"
    chmod 0600 "$TINYHELM_FLUX_JWT_PRIVATE_KEY_PATH.tmp"
    mv "$TINYHELM_FLUX_JWT_PRIVATE_KEY_PATH.tmp" "$TINYHELM_FLUX_JWT_PRIVATE_KEY_PATH"
    openssl pkey -in "$TINYHELM_FLUX_JWT_PRIVATE_KEY_PATH" -pubout -out "$TINYHELM_FLUX_JWT_PUBLIC_KEY_PATH.tmp"
    chmod 0644 "$TINYHELM_FLUX_JWT_PUBLIC_KEY_PATH.tmp"
    mv "$TINYHELM_FLUX_JWT_PUBLIC_KEY_PATH.tmp" "$TINYHELM_FLUX_JWT_PUBLIC_KEY_PATH"
fi

if ! /usr/local/bin/flux --version >/dev/null 2>&1; then
    echo "   ERROR  Flux binary cannot run in this container. Check that the installed coold nightly Flux artifact is compatible with Alpine."
    exec sleep infinity
fi

mkdir -p "$(dirname "$TINYHELM_FLUX_UNIX_SOCKET_PATH")"

echo "   INFO  Flux is enabled for role '$role', starting..."
exec /usr/local/bin/flux
